INICIO
    config <- {}
    archivo <- ABRIR "config.txt" EN MODO LECTURA
    
    PARA CADA linea EN archivo:
        clave, valor <- DIVIDIR linea POR "="
        SI "." EN valor ENTONCES
            config[clave] <- CONVERTIR_A_FLOAT(valor)
        SINO
            config[clave] <- CONVERTIR_A_ENTERO(valor)
    CERRAR archivo

    datos <- LEER_TODAS_LAS_LINEAS("clase.csv")
    
    alto <- config["alto"]
    ancho <- config["ancho"]
    max_iter <- config["max_iter"]

    img <- CREAR_IMAGEN_HSV(ancho, alto)

    // Quitar encabezados
    ELIMINAR_PRIMER_ELEMENTO(datos)

    PARA CADA dato EN datos:
        fila, columna, iteraciones <- SEPARAR dato POR ","
        
        SI iteraciones == max_iter ENTONCES
            brillo <- 0
        SINO
            brillo <- CONVERTIR_A_ENTERO((iteraciones / max_iter) * 255)
        
        COLOR <- (brillo, 255, 255)
        UBICACION <- (columna, fila)
        PONER_PIXEL(img, UBICACION, COLOR)
    
    img_rgb <- CONVERTIR_A_RGB(img)
    GUARDAR_IMAGEN(img_rgb, "mandelbrot-clase2.png")

    ESCRIBIR "DONE"
FIN
